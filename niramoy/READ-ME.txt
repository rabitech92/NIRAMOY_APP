USE niramoy;

LOAD DATA INFILE 'C:/ProgramData/MySQL/MySQL Server 8.0/Uploads/medicines.csv'
INTO TABLE medicines
FIELDS TERMINATED BY ',' OPTIONALLY ENCLOSED BY '"'
LINES TERMINATED BY '\n'

(name, category, generic_name, company_name, barcode, batch_no,
 expiry_date, cost_price, sale_price, stock, reorder_level,
